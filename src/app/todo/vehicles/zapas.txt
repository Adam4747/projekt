package com.example.demo.Spaceship;

import org.springframework.boot.CommandLineRunner;
import org.springframework.stereotype.Component;

@Component
public class SpaceshipDataInitializer implements CommandLineRunner {

    private final SpaceshipRepository repo;

    public SpaceshipDataInitializer(SpaceshipRepository repo) {
        this.repo = repo;
    }

    @Override
    public void run(String... args) {

        repo.deleteAll();
        System.out.println("INIT SPACESHIPS...");


        StarWarsSpaceship s1 = new StarWarsSpaceship();
        s1.setName("ARC-170");
        s1.setModel("Aggressive Reconnaissance-170 starfighter");
        s1.setCreator("Incom Corporation");
        s1.setPrice("155000");
        s1.setLength("14.5m");
        s1.setMaxSpeed("1000");

        StarWarsSpaceship s2 = new StarWarsSpaceship();
        s2.setName("Star Destroyer");
        s2.setModel("Imperial I-class Star Destroyer");
        s2.setCreator("Kuat Drive Yards");
        s2.setPrice("1.200.000.000");
        s2.setLength("1600m");
        s2.setMaxSpeed("975");

        StarWarsSpaceship s3 = new StarWarsSpaceship();
        s3.setName("Death Star");
        s3.setModel("DS-1 Orbital Battle Station");
        s3.setCreator("Imperial Engineering Corps");
        s3.setPrice("nie do oszacowania");
        s3.setLength("120000km");
        s3.setMaxSpeed("superlaser station");

        StarWarsSpaceship s4 = new StarWarsSpaceship();
        s4.setName("Millennium falcon");
        s4.setModel("YT-1300 light freighter");
        s4.setCreator("Corellian Engineering Corporation");
        s4.setPrice("100000");
        s4.setLength("34.75m");
        s4.setMaxSpeed("1050");

        StarWarsSpaceship s5 = new StarWarsSpaceship();
        s5.setName("TIE Advanced X1");
        s5.setModel("Twin Ion Engine Advanced x1");
        s5.setCreator("Sienar Fleet Systems");
        s5.setPrice("unknown");
        s5.setLength("9.2m");
        s5.setMaxSpeed("1200");

        StarWarsSpaceship s6 = new StarWarsSpaceship();
        s6.setName("X-Wing");
        s6.setModel("T-65 X-wing");
        s6.setCreator("Incom Corporation");
        s6.setPrice("149999");
        s6.setLength("12.5m");
        s6.setMaxSpeed("1050");

        repo.save(s1);
        repo.save(s2);
        repo.save(s3);
        repo.save(s4);
        repo.save(s5);
        repo.save(s6);
    }
}
